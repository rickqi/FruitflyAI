### 1.2 六步迁移法 (通用领域适配框架)

无论目标领域如何，以下六步迁移框架保持恒定：

| 步骤 | 操作 | 涉及Fly64模块 | 代码修改量估计 |
|------|------|-------------|--------------|
| **S1: 输入适配** | 替换SM64帧/游戏状态 → 目标域传感器数据 | `retina.py`, `model.py#encode_retina()` | 中-高(取决于传感器模态) |
| **S2: 奖励重写** | 替换位移奖励 → 目标域成功/失败指标 | `mushroom_body.py`, `gain_modulation.py` | 中(2-5个奖励信号) |
| **S3: 动作映射** | 4自由度 → 目标域动作基元(可扩展) | `model.py#step()`投票解码, MBON输出 | 低-中(替换动作名称+维度) |
| **S4: 环境对接** | mmap桥接 → 目标域控制协议 | `bridge.py`, `main.py` | 低(扩展Header) |
| **S5: 监控适配** | 仪表板指标映射 + 因果链时间轴 | `telemetry.py`, `dashboard.js` | 低(指标重命名) |
| **S6: 闭环验证** | Hybrid模式 → 逐步增加SNN占比 | `evolution_skill.py` | 低(诊断指标替换) |

### 1.3 决策仲裁链 (通用跨领域模式)

Fly64的5级决策仲裁链在所有领域中保持一致：

```
优先级: 反射 > LIF内生决策 > 多巴胺门控学习 > LLM咨询 > 局部降级
─────────────────────────────────────────────────────────────────────
Level 0 (最高): 反射注入 — 直接电流注入膜电位, 50Hz即时响应
Level 1: LIF脉冲网络 — 基于连接组的固有动力学, 持续运行
Level 2: 蘑菇体可塑性 — 多巴胺门控的联想学习, 长期自适应
Level 3: LLM教官 — 仅当内生系统陷入困境时激活(10s周期)
Level 4 (最低): 局部降级 — LLM不可用时的自治后备
```

### 2.6 关键代码修改规格

```python
# === retina.py → risk_encoder.py (新文件) ===
# 替换 SphericalRetina 为 RiskFeatureEncoder
class RiskFeatureEncoder:
    """果蝇嗅觉回路启发的风险特征编码器"""
    def __init__(self, input_dim=50, output_dim=128, seed=64):
        # 保留随机投影矩阵(128×50) 替代 128×1536
        self.projection = np.random.RandomState(seed).randn(output_dim, input_dim)
        self.projection /= np.linalg.norm(self.projection, axis=1, keepdims=True)
    
    def encode(self, risk_input):
        """risk_input: 50维稀疏风险特征向量"""
        # 5%稀疏编码(同KC编码) — 保留果蝇嗅觉编码原理
        features = self.projection @ risk_input
        top_k = int(0.05 * len(features))
        indices = np.argpartition(features, -top_k)[-top_k:]
        sparse = np.zeros_like(features)
        sparse[indices] = features[indices]
        return sparse  # 128维稀疏向量
```

```python
def encode_place_signature(self, camera_frame, imu_heading):
    """机器人的位置签名(视觉+里程计融合)"""
    # 1. 视觉特征 (128维, 零修改)
    visual_sig = self.scene_projection @ camera_frame.ravel()
    
    # 2. IMU航向 (新的融合信号)
    heading_bias = np.sin(imu_heading) * 0.1  # 周期性航向调谐
    place_sig = visual_sig + heading_bias  # 融合签名
    
    # 3. 余弦相似度回环检测 (零修改)
    best_match = self.memory_controller.query_scene_db(place_sig)
    if best_match['similarity'] > 0.85:  # 回环阈值
        self.loop_closure_detected = True
        self.estimated_position = best_match['grid_position']
    
    return place_sig
```

### 5.2 τ估计 → AEB碰撞预警 — 数学等价性

```python
# === retina.py — LC4 SectorLoom (零修改, 仅参数校准) ===

# Fly64原始τ估计:
def compute_looming(self, flow_energy, dt=0.02):
    """果蝇LC4神经元的逼近检测"""
    d_theta_dt = np.sqrt(flow_energy)  # 视场角膨胀率
    tau = 1.0 / (d_theta_dt + 1e-6)   # time-to-contact
    return tau  # 帧

# 自动驾驶AEB参数校准:
# 物理τ (秒) = tau × dt(0.02s) × 速度因子
# AEB触发阈值:
#   - FCW预警: τ < 2.0s (约前向碰撞2秒前告警)
#   - AEB部分制动: τ < 1.5s
#   - AEB全力制动: τ < 0.8s (Euro NCAP 5-star标准)
# 
# 参数校准:
#   TAU_NEAR: 0.5 (原) → 根据不同速度动态: 
#     tau_near = max(0.3, min(1.0, velocity_mps / 30.0))
#   TAU_DECELERATE: 1.5 (原) → 2.0 (AEB符合Euro NCAP)
#   TAU_SHARP_TURN: 2.5 (原) → 3.0 (更早的转向规避)

class AEBCalibration:
    """τ参数动态校准(基于车速)"""
    @staticmethod
    def get_thresholds(vehicle_speed_mps):
        speed = vehicle_speed_mps * 3.6  # m/s→km/h
        if speed < 30:  # 低速城市
            return {'aeb_full': 0.5, 'aeb_part': 1.0, 'fcw': 1.5}
        elif speed < 60:  # 中速郊区
            return {'aeb_full': 0.8, 'aeb_part': 1.5, 'fcw': 2.0}
        else:  # 高速
            return {'aeb_full': 1.2, 'aeb_part': 2.0, 'fcw': 3.0}
```

### 6.2 市场特征编码器 (替换视觉管道)

```python
# === market_encoder.py (新文件, 替换retina.py) ===
# 果蝇嗅觉编码原理 → 市场状态编码

class MarketFeatureEncoder:
    """
    价格→脉冲编码器(果蝇嗅觉回路启发的市场特征提取)
    
    核心原理:
    - 价格变化 → ON/OFF通道(类似视觉ON/OFF帧差)
    - 成交量变化 → 运动检测(类似EMD)
    - 波动率加速 → τ估计(类似LC4逼近检测)
    """
    
    def __init__(self, lookback=20, n_features=128):
        self.lookback = lookback
        self.price_buffer = np.zeros(lookback)
        self.volume_buffer = np.zeros(lookback)
        self.feature_projection = np.random.RandomState(64).randn(
            n_features, lookback * 5  # 5个通道×lookback
        )
        self.feature_projection /= np.linalg.norm(
            self.feature_projection, axis=1, keepdims=True
        )
    
    def encode_market(self, ohlcv):
        """将OHLCV行情数据编码为128维市场特征向量"""
        close, volume = ohlcv[:, 3], ohlcv[:, 4]
        
        # === ON通道: 价格上涨(类似视觉ON瞬变) ===
        price_up = np.maximum(0, np.diff(close) / close[:-1])
        price_up_padded = np.pad(price_up, (1, 0))
        
        # === OFF通道: 价格下跌(类似视觉OFF瞬变) ===
        price_down = np.maximum(0, -np.diff(close) / close[:-1])
        price_down_padded = np.pad(price_down, (1, 0))
        
        # === 成交量通道: 交易活跃度(类似运动检测) ===
        vol_change = np.diff(volume) / (volume[:-1] + 1e-8)
        vol_norm = np.clip(vol_change, -1, 1)
        vol_padded = np.pad(vol_norm, (1, 0))
        
        # === 波动率τ: volatility-to-contact (类似LC4 looming) ===
        returns = np.diff(close) / close[:-1]
        vol_accel = np.diff(returns)  # 价格变化的加速度
        vol_tau = 1.0 / (np.abs(vol_accel) + 1e-8)
        vol_tau_clipped = np.clip(vol_tau, 0.1, 100.0) / 100.0
        vol_tau_padded = np.pad(vol_tau_clipped, (2, 0))[:len(close)]
        
        # === 价差通道: 买卖力量对比(类似EMD方向) ===
        if hasattr(ohlcv, 'shape') and ohlcv.shape[1] >= 5:
            spread = np.abs(close - np.mean(ohlcv[:, :4], axis=1)) / close
            spread_norm = np.clip(spread, 0, 0.1) / 0.1
        else:
            spread_norm = np.zeros_like(close)
        
        # 5通道特征向量构建
        features = np.column_stack([
            price_up_padded[-self.lookback:],
            price_down_padded[-self.lookback:],
            vol_padded[-self.lookback:],
            vol_tau_padded[-self.lookback:],
            spread_norm[-self.lookback:]
        ]).ravel()
        
        # 128维随机投影(保留果蝇嗅觉编码) → 保持5%稀疏
        market_sig = self.feature_projection @ features
        top_k = int(0.05 * len(market_sig))
        indices = np.argpartition(np.abs(market_sig), -top_k)[-top_k:]
        sparse = np.zeros_like(market_sig)
        sparse[indices] = market_sig[indices]
        
        return sparse  # 128维稀疏市场特征向量
```

```python
# === mushroom_body.py — 交易奖励信号重写 ===
# 替换SM64位移奖励为交易PnL奖励

class TradingDopamineSources:
    """量化交易的多巴胺奖励源(替换游戏奖励)"""
    
    @staticmethod
    def compute_rewards(portfolio_state, trade_log):
        """
        计算交易多巴胺信号
        
        输入: 投资组合状态(仓位/PnL/持仓时间)
        输出: 多巴胺奖励字典
        """
        rewards = {}
        
        # 1. 主奖励: PnL变化 (交易盈亏)
        pnl_change = portfolio_state['total_pnl'] - portfolio_state['prev_pnl']
        rewards['reward_signal'] = np.tanh(pnl_change * 10.0)  # 归一化[-1, 1]
        
        # 2. 持有奖励: 趋势跟随成功(仓位方向与市场一致)
        if portfolio_state['position'] != 0:
            pos_direction = np.sign(portfolio_state['position'])
            price_change = portfolio_state['price_change']
            trend_reward = pos_direction * price_change
            rewards['movement_reward'] = np.clip(trend_reward, -0.5, 0.5)
        else:
            rewards['movement_reward'] = 0.0
        
        # 3. 挫折信号: 亏损/回撤惩罚
        drawdown = portfolio_state['max_drawdown']
        if drawdown > 0.05:  # 回撤>5%
            rewards['_pending_dopamine'] = -min(drawdown * 2.0, 0.8)
        elif portfolio_state['loss_streak'] > 3:  # 连续亏损
            rewards['_pending_dopamine'] = -0.3
        else:
            rewards['_pending_dopamine'] = 0.0
        
        # 4. 交易频率惩罚: 过度交易抑制
        trade_freq = portfolio_state['trades_last_hour']
        if trade_freq > 10:  # 每小时超过10笔
            rewards['_pending_dopamine'] += -0.1 * (trade_freq - 10)
        
        return rewards
```

### 8.2 状态编码器(替换视觉管道)

```python
# === arm_state_encoder.py (新文件) ===

class ArmStateEncoder:
    """
    机械臂状态编码器(果蝇体感系统启发)
    
    输入: 关节状态 + 末端执行器 + 视觉反馈 + 目标信息
    输出: 128维稀疏特征向量(与MushroomBody兼容)
    """
    
    def __init__(self, n_joints=6, seed=64):
        self.n_joints = n_joints
        
        # 状态维度:
        # 6关节角度 + 6关节速度 + 3末端位置 + 3末端方向
        # + 1抓取力 + 3目标位置 + 128视觉签名
        state_dim = 6 + 6 + 3 + 3 + 1 + 3 + 128
        
        # 随机投影(保留种子64 → 可复现)
        self.projection = np.random.RandomState(seed).randn(128, state_dim)
        self.projection /= np.linalg.norm(self.projection, axis=1, keepdims=True)
    
    def encode_state(self, joint_angles, joint_velocities, 
                      end_effector_pos, end_effector_orient,
                      grip_force, target_pos, visual_sig):
        """
        编码机械臂完整状态为128维特征向量
        """
        # 1. 归一化关节角度(0-2π) → [-1, 1]
        angles_norm = np.sin(joint_angles)  # 角度周期性编码
        
        # 2. 归一化速度
        vel_norm = np.tanh(joint_velocities * 0.1)
        
        # 3. 末端位置(米) → 相对目标
        rel_pos = target_pos - end_effector_pos
        rel_pos_norm = np.tanh(rel_pos * 5.0)  # 1m范围归一化
        
        # 4. 抓取力(牛顿)
        force_norm = np.tanh(grip_force * 0.5)
        
        # 5. 128维视觉签名(来自视觉管道)
        
        # 组合状态向量
        state = np.concatenate([
            angles_norm,       # 6
            vel_norm,          # 6
            rel_pos_norm,      # 3
            np.tanh(end_effector_orient * 0.5),  # 3
            [force_norm],      # 1
            np.tanh(target_pos * 0.5),  # 3
            visual_sig         # 128
        ])
        
        # 随机投影 + 5%稀疏(保持KC编码)
        features = self.projection @ state
        top_k = int(0.05 * 128)
        indices = np.argpartition(np.abs(features), -top_k)[-top_k:]
        sparse = np.zeros(128)
        sparse[indices] = features[indices]
        
        return sparse
```

### 8.4 多巴胺奖励函数

```python
class ArmDopamineRewards:
    """机械臂操控的多巴胺奖励"""
    
    @staticmethod
    def compute(prev_state, current_state, target):
        rewards = {}
        prev_pos = prev_state['end_effector']
        curr_pos = current_state['end_effector']
        
        # 1. 主奖励: 到目标的距离变化
        prev_dist = np.linalg.norm(prev_pos - target)
        curr_dist = np.linalg.norm(curr_pos - target)
        dist_change = prev_dist - curr_dist  # 正=接近目标
        
        rewards['reward_signal'] = np.tanh(dist_change * 5.0)  # 归一化[-1,1]
        
        # 2. 成功奖励: 抓取到目标物体
        if current_state['grasp_success']:
            rewards['movement_reward'] = 0.8  # 大额奖励
        
        # 3. 挫折信号
        if current_state['collision']:
            rewards['_pending_dopamine'] = -0.6  # 碰撞惩罚
        elif current_state['object_dropped']:
            rewards['_pending_dopamine'] = -0.5  # 掉落惩罚
        elif prev_dist < 0.05 and curr_dist > 0.05:
            rewards['_pending_dopamine'] = -0.3  # 靠近目标后远离=沮丧
        else:
            rewards['_pending_dopamine'] = 0.0
        
        # 4. 能量效率惩罚(减少不必要的力矩)
        torque_penalty = np.mean(np.abs(current_state['joint_torques'])) * 0.1
        rewards['_pending_dopamine'] -= torque_penalty
        
        return rewards
```
